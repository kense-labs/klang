decorator Schema(label: const u8[:]) on struct;
decorator Choice() on enum;
decorator Column(index: u8) on field;

@Schema("row")
root struct Row<T> {
    @Column(1) let data: T,
}

@Choice
enum Event {
    Data { @Column(2) data: i64 },
    End,
}

fn main() {
    let schema = decorator_metadata<Schema, Row>();
    let column = decorator_metadata<Column, Event.Data>(data);
}
